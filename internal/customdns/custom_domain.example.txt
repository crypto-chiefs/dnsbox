# Custom DNS records for DNSBox. Configure DNSBOX_RECORDS_FILE to load a file.
# Format: <fully qualified name> <type> <value>
# Every name must be inside DNSBOX_DOMAIN. Supported: A, AAAA, CNAME, TXT, MX.
# Example zone: lab.example.com
lab.example.com A 192.168.1.10
landing.lab.example.com A 192.168.1.100
landing.lab.example.com AAAA fd00::100
landing.lab.example.com TXT "Hello world"
client.lab.example.com CNAME landing.lab.example.com.
lab.example.com MX 10 mail.lab.example.com.
